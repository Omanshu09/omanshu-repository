  // Scroll reveal
  const items = document.querySelectorAll('.reveal');
  const io = new IntersectionObserver((entries)=>{
    entries.forEach(e=>{ if(e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target);} });
  }, { threshold:0.15 });
  items.forEach(i=>io.observe(i));

  // Mobile nav toggle
  const navToggle = document.getElementById('navToggle');
  const navLinks = document.getElementById('navLinks');
  navToggle.addEventListener('click', ()=>{
    navToggle.classList.toggle('open');
    navLinks.classList.toggle('open');
  });
  navLinks.querySelectorAll('a').forEach(a=>{
    a.addEventListener('click', ()=>{
      navToggle.classList.remove('open');
      navLinks.classList.remove('open');
    });
  });

  // Scroll progress bar
  const progressBar = document.getElementById('progressBar');
  function updateProgress(){
    const h = document.documentElement;
    const scrolled = (h.scrollTop) / (h.scrollHeight - h.clientHeight) * 100;
    progressBar.style.width = scrolled + '%';
  }
  window.addEventListener('scroll', updateProgress, { passive:true });
  updateProgress();

  // Back to top button
  const backTop = document.getElementById('backTop');
  window.addEventListener('scroll', ()=>{
    backTop.classList.toggle('show', window.scrollY > 700);
  }, { passive:true });
  backTop.addEventListener('click', ()=> window.scrollTo({ top:0, behavior:'smooth' }));

  // Scroll-progress drawing line (vanilla-JS version of the
  // Framer Motion "stroke follows scroll" reference you shared)
  const scrollPath = document.getElementById('scrollPath');
  const pathEl = scrollPath.querySelector('path');
  let pathLength = pathEl.getTotalLength();

  function sizeScrollPath(){
    const docHeight = document.documentElement.scrollHeight;
    scrollPath.style.height = docHeight + 'px';
    pathLength = pathEl.getTotalLength();
    pathEl.style.strokeDasharray = pathLength;
    updateScrollPath();
  }

  function updateScrollPath(){
    const scrollTop = window.scrollY;
    const maxScroll = document.documentElement.scrollHeight - window.innerHeight;
    const progress = maxScroll > 0 ? Math.min(1, scrollTop / maxScroll) : 0;
    pathEl.style.strokeDashoffset = pathLength * (1 - progress);
  }

  window.addEventListener('scroll', updateScrollPath, { passive:true });
  window.addEventListener('resize', sizeScrollPath);
  sizeScrollPath();

  const sectionIds = ['about','skills','experience','projects','education','contact'];
  const sections = sectionIds.map(id=>document.getElementById(id)).filter(Boolean);
  const navAnchors = document.querySelectorAll('.navlinks a, .side-nav a');
  const spy = new IntersectionObserver((entries)=>{
    entries.forEach(entry=>{
      if(entry.isIntersecting){
        const id = entry.target.getAttribute('id');
        navAnchors.forEach(a=>{
          a.classList.toggle('active', a.getAttribute('href') === '#'+id);
        });
      }
    });
  }, { rootMargin:'-40% 0px -50% 0px', threshold:0 });
  sections.forEach(s=>spy.observe(s));

  // Expandable project cards
  document.querySelectorAll('.proj-card').forEach(card=>{
    const hint = document.createElement('div');
    hint.className = 'proj-hint';
    hint.textContent = 'Click to collapse ▴';
    card.appendChild(hint);
    card.addEventListener('click', ()=>{
      const collapsedNow = card.classList.toggle('collapsed');
      hint.textContent = collapsedNow ? 'Click to expand ▾' : 'Click to collapse ▴';
    });
  });
</script>
